Fantasy Baseball WAR Project

In this personal project, I wrote code in R to calculate the value of player, given a fantasy baseball league’s scoring system. WAR (Wins Above Replacement) is a statistic that Major League Baseball teams use to calculate the value a player adds over a replacement-level player at the same position. I applied the same concept to fantasy baseball in this project, where I identified:

  1. Replacement-level players at every position: I found the level of replacement by looking at which players were on a team, and which players were listed as ‘free agents.’ Players listed as free agents may be added by any team, which aligns with the definition of replacement-level.

  2. Points needed for a win: To win a game in fantasy baseball, a team must score more points than their opponent. To find the amount of points needed to win a matchup, I compiled a list of every matchup’s winners and averaged their point totals. In theory, this stat describes ‘what it takes’ to win a game, on average. This value ended up being 447.29 points per game

Finally, to calculate the WAR, I started by determining how many wins a player creates, which I did by dividing their total points by the team points needed for a win. To find how many wins their replacement generated, I used the same process. The difference between the two values describes the “Wins Above Replacement” (or WAR) for a player. The equation is shown below:

WAR = (pts / 447.29) - (catcher_rep / 447.29)

Code Sample for Hitters

catcher_hit <- mlb_stats(stat_type = 'season',          #pulls data using mlb_stats package
  position = "C",                                       #selects only Catchers
  player_pool = 'all',
  stat_group = 'hitting',                               #pulls only hitting statistics
  season = Season,                                      
  limit = 50) |>                                        #takes only the top 50 catchers
  mutate(G = games_played,                              #renames stats to abbreviations
         R = runs, 
         HR = home_runs,
         K = strike_outs,
         BB = base_on_balls,
         IBB = intentional_walks,
         H = hits,
         HBP = hit_by_pitch,
         AB = at_bats,
         SB = stolen_bases,
         GIDP = ground_into_double_play,
         PA = plate_appearances,
         TB = total_bases,
         RBI = rbi,
         BABIP = babip,
         name = player_full_name,
         Pos = "C") |>
  mutate(pts = (R*1 + RBI*1 + H*1 + TB*1 + BB*2 + IBB*1 - K*1 - GIDP*1 + SB*2)) |>   #calculates points for a player, given the league scoring system for each hitting statistic
  select(name, pts, Pos, G, PA, AB, H, TB, RBI, R, HR, BB, IBB, K, SB, GIDP, HBP, BABIP)
         
catcher_field <- mlb_stats(stat_type = 'season',
  position = "C",
  player_pool = 'all',
  stat_group = 'fielding',                            #selects fielding statistics
  season = Season, 
  limit = 750) |> 
  mutate(name = player_full_name,                     #renames columns succinctly
         E = errors) |>
  group_by(name) |>
  summarize(E = sum(E, na.rm = TRUE)) |>
  select(name, E)                                     #keeps only player name and Errors

catchers <- catcher_hit |>                            #combines data frames
  left_join(catcher_field, by = "name") |>
  mutate(pts = (pts - E),                        #calculates total points with fielding errors
         ppg = pts/G) |>                         #calculates points per game
  select(name, pts, ppg, Pos, G, PA, AB, H, TB, RBI, R, HR, BB, IBB, K, SB, GIDP, HBP, BABIP, E) |>
  arrange(desc(pts))                                  #arranges column by points
    
    
catcher_rep <- catchers |>                            #saves replacement-level catcher
  ungroup() |>
  arrange(desc(pts), desc(ppg)) |>
  distinct() |>
  slice(15) |>                                       #15th best catcher is 'replacement-level'
  pull(pts)
  
catchers <- catchers |>                              #edits the 'catchers' data frame
  ungroup() |>
  arrange(desc(pts), desc(ppg)) |>
  mutate(WAR = round((pts / 447.29) - (catcher_rep / 447.29)), 3) |>             #finds a player's WAR in a year, given that 447.29 points creates a win
  select(name, WAR, pts, ppg, Pos, G, PA, AB, H, TB, RBI, R, HR, BB, IBB, K, SB, GIDP, HBP, BABIP, E) |>
  distinct()

rm(catcher_field, catcher_hit)                #cleans environment in R

Code Sample for Pitchers

convert_innings <- function(ip) {
  whole <- floor(ip)                    #finds whole number in IP statistic
  decimal <- ip - whole                 #finds the decimal in the IP statistic
  if (decimal >= 0.3) return(NA_real_)  #avoids any bad conversions
  return(whole + (decimal * 10) / 3)}   #recalculates decimal to be more statistically accurate

starting_pitchers <- mlb_stats(stat_type = 'season',
  position = "P",                        #finds only pitchers
  player_pool = 'all',
  stat_group = 'pitching',
  season = Season, 
  limit = 750) |>
  mutate(G = games_played,
         GS = games_started,
         GO = ground_outs,
         FO = air_outs,
         HR = home_runs,
         K = strike_outs,
         BB = base_on_balls,
         IBB = intentional_walks,
         H = hits,
         HBP = hit_by_pitch,
         AB = at_bats,
         GIDP = ground_into_double_play,
         Pitches = number_of_pitches,
         IP_raw = as.numeric(innings_pitched),
         SV = saves,
         HD = holds,
         BS = blown_saves,
         ER = earned_runs,
         `Strike %` = strike_percentage,
         B = balks,
         PKO = pickoffs, 
         name = player_full_name,
         Pos = "SP") |>
  mutate(IP = sapply(IP_raw, convert_innings)) |>     #uses function to convert IP
  mutate(pts = (IP*4.5 - H*1 - ER*3 - BB*1 + K*1 - B*1 + PKO*1 + SV*6 + HD*5 - BS*3),
         ppg = pts/G) |>              #calculates points using league's scoring system
  select(name, pts, ppg, Pos, G, GS, IP, K, BB, H, ER, SV, HD, BS, B, PKO, `Strike %`, Pitches, AB, GIDP, HBP, IBB, GO, FO) |>
  filter((Pitches/G) >= 30) |>        #only takes pitchers that tend to throw more than 30 pitchers per game (relief pitchers will throw under 30, starters will throw over 30)
  arrange(desc(pts))
    
starting_pitcher_rep <- starting_pitchers |>        #saves replacement-level starting pitcher

  arrange(desc(pts), desc(ppg)) |>
  distinct() |>
  slice(92) |>                              #92nd best starting pitcher is 'replacement-level'
  pull(pts)
